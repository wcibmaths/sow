// NOW  = the lesson currently marked In progress (first one, in scheme order).
// NEXT = the first Not-started lesson AFTER it. Deliberately not the earliest gap
//        anywhere in the scheme: if someone forgot to tick lesson 4 but is now
//        teaching lesson 10, the next thing they teach is 11, not 4.
function getClassPosition(yg,set){
  const sow=sowFor(yg,set);
  if(!sow) return null;
  const teachable=getTeachable(sow);
  if(!teachable.length) return null;
  const statuses=teachable.map(l=>getStatus(l.id,set));
  const curIdx=statuses.indexOf('In progress');
  let nextIdx=-1;
  for(let i=(curIdx>=0?curIdx+1:0);i<teachable.length;i++){
    if(statuses[i]==='Not started'){ nextIdx=i; break; }
  }
  let lastDoneIdx=-1;
  for(let i=teachable.length-1;i>=0;i--){ if(statuses[i]==='Done'){ lastDoneIdx=i; break; } }
  return {
    current : curIdx>=0      ? teachable[curIdx]      : null,
    next    : nextIdx>=0     ? teachable[nextIdx]     : null,
    lastDone: lastDoneIdx>=0 ? teachable[lastDoneIdx] : null,
    complete: curIdx<0 && nextIdx<0,
    started : curIdx>=0 || lastDoneIdx>=0
  };
}
// Unit is numeric in most schemes but free text in a few (Y11 accelerated uses
// "Course opening"), and some unit names already carry their own number
// ("0. Start of Year"), so only prefix "Unit n" when it is a real number.
function posUnit(l){
  if(!l) return '';
  const name=String(l.unitName||'').trim();
  const u=l.unit;
  const n=String(u==null?'':u).trim();
  if(!/^\d+$/.test(n) || Number(n)<=0) return name;
  if(name.indexOf(n)===0) return name;
  return `Unit ${n} · ${name}`;
}
// KS3 rows already begin with their own number ("1.1 Averages and range"), and
// some rows use a marker instead ("Revision", "★") which would read "LRevision —".
function posLesson(l){
  if(!l) return '';
  const name=String(l.lessonName||'').trim();
  const no=String(l.lessonNo||'').trim();
  if(!no || name.indexOf(no)===0) return name;
  if(!/^\d/.test(no)) return name;
  return `L${no} — ${name}`;
}
// '22-26 Sep 2026' -> '22-26 Sep'. The year is noise when every card shares it.
function posWhen(l){
  if(!l||!l.dates) return '';
  return String(l.dates).replace(/\s*\b(20\d\d)\b\s*$/,'').trim();
}
