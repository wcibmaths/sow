function buildAssessmentUrl(yg,unit){
  const yearPath=ASSESSMENT_YEAR_PATH[yg];
  const folder=(ASSESSMENT_UNIT_FOLDER[yg]||{})[unit];
  if(!yearPath || !folder) return '';
  return `${ASSESSMENT_ROOT}/${yearPath}/${folder}/Unit_${unit}_Assessment?csf=1&web=1`;
}
const ASSESSMENT_FOLDERS_BY_YG = (()=>{
  const out={};
  Object.keys(ASSESSMENT_UNIT_FOLDER).forEach(yg=>{
    const y=Number(yg); out[y]={};
    Object.keys(ASSESSMENT_UNIT_FOLDER[y]).forEach(unit=>{
      out[y][unit]=buildAssessmentUrl(y,unit);
    });
  });
  return out;
})();

// Accelerated sets sit on the scheme of work of the year above, so their unit
// tests live in that year's folder rather than their own.
const ACCELERATED_FOLDER_YEAR = { '8|Ma1':9, '9|Ma1':10 };

// The SharePoint unit-test folder for an assessment, or '' when there isn't one.
function assessmentFolderUrl(yg,set,unit){
  if(unit===null || unit===undefined || unit==='') return '';
  const y=Number(yg);
  if(y===12) return (set==='Ma1P12'||set==='Ma2P12')
    ? (Y12_P12_ASSESSMENT_FOLDERS[unit]||'') : '';
  const lookupYear = ACCELERATED_FOLDER_YEAR[`${y}|${set}`] || y;
  const map=ASSESSMENT_FOLDERS_BY_YG[lookupYear];
  return (map && map[unit]) || '';
}
